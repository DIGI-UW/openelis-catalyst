       test_name,
       count(*) AS result_count,
       percentile_cont(0.5) WITHIN GROUP (
         ORDER BY result_value) AS median_value
FROM analytics.lab_result_fact_v1
WHERE observed_at >= :since_date
GROUP BY 1, 2
ORDER BY 1
